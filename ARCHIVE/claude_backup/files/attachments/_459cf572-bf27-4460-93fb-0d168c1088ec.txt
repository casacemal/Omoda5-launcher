# 🔍 3. PRIVILEGE RESOLVER (CORE LOGIC)

```kotlin id="p3"
class PrivilegeResolver(
    private val context: Context
) {

    fun resolve(): PrivilegeMode {

        return when {

            hasCarServicePermission() ->
                PrivilegeMode.CAR_SERVICE_ACCESS

            isSystemApp() ->
                PrivilegeMode.SYSTEM_PRIVILEGED

            else ->
                PrivilegeMode.LIMITED_FALLBACK
        }
    }

    private fun hasCarServicePermission(): Boolean {
        return context.checkSelfPermission(
            "android.car.permission.CAR_SPEED"
        ) == PackageManager.PERMISSION_GRANTED
    }

    private fun isSystemApp(): Boolean {
        return (context.applicationInfo.flags and
                ApplicationInfo.FLAG_SYSTEM) != 0
    }
}
```

class VehicleDataRouter(
    private val resolver: PrivilegeResolver,
    private val car: CarSource,
    private val dumpsys: DumpsysSource

    fun getSpeed(): Float {

        return when (resolver.resolve()) {

            PrivilegeMode.SYSTEM_PRIVILEGED ->
                car.getSpeed()

            PrivilegeMode.CAR_SERVICE_ACCESS ->
                car.getSpeed()

            PrivilegeMode.LIMITED_FALLBACK ->

        }
    }
}

# ⚙️ 5. FALLBACK STRATEGY (REAL WORLD SAFE METHOD)

## dumpsys parsing (existing system)

```kotlin id="p5"
class DumpsysSource {

    fun getSpeed(): Float {

        val output = Runtime.getRuntime()
            .exec("dumpsys car_service")
            .inputStream.bufferedReader()
            .readText()

        return output.lines()
            .firstOrNull { it.contains("speed") }
            ?.filter { it.isDigit() || it == '.' }
            ?.toFloatOrNull() ?: 0f
    }
}
```

class CarSource(
    private val carPropertyManager: CarPropertyManager
) {

    fun getSpeed(): Float {

        return try {
            carPropertyManager.getProperty(
                Float::class.java,
                VehiclePropertyIds.PERF_VEHICLE_SPEED,
                0
            ).value
        } catch (e: Exception) {
            0f
        }
    }
}
