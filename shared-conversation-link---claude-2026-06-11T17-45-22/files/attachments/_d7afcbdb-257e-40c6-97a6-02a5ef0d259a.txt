# 🚗 4. VEHICLE LAYER (HYBRID)

```kotlin id="x4"
class VehicleLayer(
    private val dumpsys: DumpsysReader,
    private val car: CarSource
) {

    fun getSpeed(): Float {
        return try {
            car.getSpeed()
        } catch (e: Exception) {
            dumpsys.getSpeed()
        }
    }
}
```

# 🪟 7. OVERLAY SERVICE

```kotlin id="x7"
class OverlayService : Service() {

    override fun onCreate() {

        val wm = getSystemService(WINDOW_SERVICE) as WindowManager

        val view = View(this)

        val params = WindowManager.LayoutParams(
            300, 300,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE,
            PixelFormat.TRANSLUCENT
        )

        wm.addView(view, params)
    }

    override fun onBind(intent: Intent?) = null
}
```

# 🚗 8. MAIN ASSISTANT SERVICE

```kotlin id="x8"
class AssistantService : Service() {

    private val scope = CoroutineScope(Dispatchers.Default)

    override fun onCreate() {

        val vehicle = VehicleLayer(DumpsysReader(), CarSource())
        val audio = AudioEngine()
        val hermes = HermesClient()

        hermes.connect()

        scope.launch {

            while (true) {

                val speed = vehicle.getSpeed()

                audio.duck(if (speed > 80) 0.2f else 1f)

                if (speed > 90) {
                    hermes.send("Speed warning")
                }

                delay(1000)
            }
        }
    }

    override fun onBind(intent: Intent?) = null
}
```

# 🔌 10. HERMES BACKEND (PYTHON WS + TTS)

## backend-hermes/main.py

```python id="x10"
from fastapi import FastAPI, WebSocket
import asyncio

app = FastAPI()

@app.websocket("/ws")
async def ws(websocket: WebSocket):
    await websocket.accept()

    while True:
        data = await websocket.receive_json()

        text = data["text"]

        response = f"AI: {text}"

        await websocket.send_json({
            "type": "text",
            "msg": response
        })

        await websocket.send_json({
            "type": "audio_chunk",
            "data": "BASE64_PCM"
        })
```
